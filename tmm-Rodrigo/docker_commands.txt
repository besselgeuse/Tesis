docker images: "muestra las imagenes"
docker pull imagen:"version": "descarga la imagen del docker"
docker image rm imagen:"version" : "elimina la imagen del docker"
docker run -p8000:8000 --name mi_app -d imagen:"version": "ejecuta la imagen del docker en segundo plano"
docker stop nombre_contenedor : "detiene la imagen del docker"
docker create imagen:"version" : "crea la imagen del docker pero no la ejecuta"
docker start nombre_contenedor : "inicia la imagen del docker"
docker create --name mi_app imagen:"version": "crea la imagen del docker pero no la ejecuta"
docker create -p8000:8000 --name mi_app imagen:"version": "crea la imagen del docker pero no la ejecuta y mapea el puerto 8000 del contenedor al puerto 8000 del host"
docker ps : "muestra los contenedores en ejecucion"
docker ps -a: "muestra todos los contenedores"
docker container rm nombre_contenedor : "elimina el contenedor"
